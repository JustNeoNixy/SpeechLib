cmake_minimum_required(VERSION 3.24)
project(speechlib_gpu LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
foreach(FLAG AVX AVX2 FMA F16C BMI2 OPENMP)
    set(GGML_${FLAG} OFF CACHE BOOL "" FORCE)
endforeach()
if(APPLE)
    set(GGML_METAL ON CACHE BOOL "" FORCE)
    set(GGML_METAL_EMBED_LIBRARY ON CACHE BOOL "" FORCE)
    set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
    set(CMAKE_INSTALL_RPATH "@loader_path")
else()
    set(GGML_METAL OFF CACHE BOOL "" FORCE)
    set(GGML_VULKAN ON CACHE BOOL "" FORCE)
    set(CMAKE_INSTALL_RPATH "$ORIGIN")
endif()
set(CMAKE_BUILD_WITH_INSTALL_RPATH TRUE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_CURL OFF CACHE BOOL "" FORCE)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
add_subdirectory("${WHISPER_SOURCE}" whisper)
foreach(LIBRARY whisper ggml ggml-base ggml-cpu ggml-vulkan ggml-metal)
    if(TARGET ${LIBRARY})
        set_target_properties(${LIBRARY} PROPERTIES OUTPUT_NAME "speechlib-${LIBRARY}" VERSION "" SOVERSION "")
    endif()
endforeach()
add_library(speechlib-gpu SHARED bridge.cpp)
target_link_libraries(speechlib-gpu PRIVATE whisper ggml)
set_target_properties(speechlib-gpu PROPERTIES PREFIX "")
if(MSVC)
    target_compile_options(speechlib-gpu PRIVATE /utf-8 /d2FH4-)
endif()
